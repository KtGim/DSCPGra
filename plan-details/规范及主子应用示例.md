
## 组件开发规范
- 简单组件（基础组件）
	```text
		src/components/ButtonGroup/
		├── index.vue      # Props、状态、computed、事件和生命周期
		├── const.js       # 有稳定常量时创建, 枚举，模板字符串等常量
		├── helper.js      # 有纯转换逻辑时创建, (utils) 工具函数
		└── readme.md      # Props、Events、Slots、ref API 和示例, 主要是说明函数用途，                                 # ai 可以识别这些 api 以便更高效的帮助我们进行开发任务
	```

- 后续计划
	- 后续业务开发和组件开发均会加入 lint 校验，严格限制index.vue 的行数（1000行，或者500行），lint 不通过无法 commit ，buid，无法上线提测。要求与主文件无关的逻辑尽量拆除去
	- 组件开发尽量要求自身状态高度内聚，不把逻辑暴露给外层组件处理，外部组件只能通过 props 变化修改子组件
	- 无特殊情况，组件样式不在父组件里，通过 !import, v-deep 等方式强制修改，尽量收敛到各自组件内部
- 成果展示：物流追踪页面目前采用上述方式开发，效果如下
	-  src/views/logistics/tracking/index.vue
		-  index.vue 总行数 525行。模板56行，js 420行，样式50行左右
		- 表单和列表内容高度收敛，script 中只包含列表操作
		- 列表cell 时组件时的操作，props可以接受 事件，通过事件把组件的数据传回到主页面中
	```vue
	<template>
	  <div class="app-container logistics-tracking">
	    <card-list :datasource="statusCards" type="view" aria-label="物流状态统计" />
	
	    <search-table
	      ref="searchTable"
	      class="tracking-search-table"
	      :model="queryParams"
	      :form-items="queryFormItems"
	      :columns="tableColumns"
	      :api="tableApi"
	      :form-props="searchFormProps"
	      :table-props="searchTableProps"
	      :pagination-config="searchTablePaginationConfig"
	      :slot-config="searchTableSlotConfig"
	      :query-builder="buildTableQuery"
	      :query-validator="validateSearch"
	      :auto-search="false"
	      :total-api="totalApi"
	      @search="handleSearch"
	      @reset="handleSearchReset"
	      @table-action-click="handleTableActionClick"
	      @selection-change="handleSelectionChange"
	      @load-success="handleTableLoadSuccess"
	      @load-error="handleTableLoadError"
	    />
	
	    <tag-management-dialog
	      ref="tagManagementDialog"
	      @confirm="reloadTable"
	      @change="reloadTable"
	    />
	
	    <remove-tag-dialog
	      ref="removeTagDialog"
	      @confirm="reloadTable"
	    />
	
	    <remark-history-drawer
	      ref="remarkHistoryDrawer"
	      @add="openSingleRemarkDialog"
	    />
	
	    <remark-dialog
	      ref="remarkDialog"
	      @confirm="handleRemarkConfirm"
	    />
	
	    <remark-import-dialog
	      ref="remarkImportDialog"
	      @confirm="reloadTable"
	    />
	
	  </div>
	</template>
	
	<script>
		import { CardList, SearchTable } from '@/components'
		import {
		  exportLogisticsTrackingApi,
		  getLogisticsTrackingPageApi,
		  getLogisticsTrackingTotalApi,
		  removeLogisticsTrackingOrderTagApi,
		  getLogisticsTrackingStatisticsApi,
		  pullLogisticsTrace
		} from '@/api/logisticsManage/logisticsTracking'
		import {
		  QUERY_CONDITION_FIELDS,
		  SEARCH_TABLE_PAGINATION_CONFIG,
		  TABLE_PAGE_SIZE,
		  TABLE_PAGE_SIZES,
		  TABLE_PAGINATION_PROPS,
		  TABLE_ROW_KEY,
		  TABLE_VIRTUAL_SCROLL_PROPS
		} from './const'
		import {
		  buildLogisticsTrackingQuery,
		  createQueryFormItems,
		  createQueryParams,
		  createSearchTableSlotConfig,
		  createStatisticsCards,
		  formatStatisticsLastStatTime,
		  findInvalidDateRange,
		  hasFilledField,
		  isStrictlyOrderedDateRange
		} from './helper'
		import { collectValues } from '@/components/utils'
		import { createTableColumns } from './tableColumns'
		import {
		  RemarkDialog,
		  RemarkHistoryDrawer,
		  RemarkImportDialog,
		  RemoveTagDialog,
		  TagManagementDialog
		} from './components'
		
		export default {
		  name: 'LogisticsTracking',
		  components: {
		    CardList,
		    SearchTable,
		    RemarkDialog,
		    RemarkHistoryDrawer,
		    RemarkImportDialog,
		    RemoveTagDialog,
		    TagManagementDialog
		  },
		  data() {
		    const queryParams = createQueryParams()
		    return {
		      tableApi: getLogisticsTrackingPageApi,
		      tableColumns: createTableColumns({
		        onDeleteTag: (row, tag) => this.handleRowTagClick(row, tag),
		        onOpenRemarkHistory: row => this.openRemarkHistory(row),
		        onOpenTracking: row => this.openTracking(row)
		      }),
		      tablePageSize: TABLE_PAGE_SIZE,
		      tablePaginationProps: TABLE_PAGINATION_PROPS,
		      tableRowKey: TABLE_ROW_KEY,
		      tableVirtualScrollProps: TABLE_VIRTUAL_SCROLL_PROPS,
		      searchTablePaginationConfig: SEARCH_TABLE_PAGINATION_CONFIG,
		      previousQuickFilters: [],
		      selectedRows: [],
		      exportLoading: false,
		      queryParams,
		      totalApi: {
		        api: getLogisticsTrackingTotalApi,
		        params: buildLogisticsTrackingQuery(queryParams),
		        dataPath: 'data'
		      },
		      statusCards: createStatisticsCards(),
		      lastStatTime: ''
		    }
		  },
		  created() {
		    this.loadStatistics()
		  },
		  mounted() {
		    this.reloadTable()
		  },
		  computed: {
		    queryFormItems() {
		      const { trackStatusList } = this.queryParams
		      return createQueryFormItems(
		        this.handleQuery,
		        this.handleDateRangeChange,
		        this.handleQuickFilterChange,
		        trackStatusList
		      )
		    },
		    hasQueryCondition() {
		      return hasFilledField(this.queryParams, QUERY_CONDITION_FIELDS)
		    },
		    canExportOrPull() {
		      return this.hasQueryCondition || this.selectedRows.length > 0
		    },
		    searchTableSlotConfig() {
		      return createSearchTableSlotConfig({
		        canExport: this.canExportOrPull,
		        exportLoading: this.exportLoading,
		        lastStatTime: this.lastStatTime
		      })
		    },
		    searchFormProps() {
		      return {
		        labelPosition: 'left',
		        labelWidth: '80px',
		        size: 'mini',
		        class: 'compact-query-form'
		      }
		    },
		    searchTableProps() {
		      return {
		        fullscreen: true,
		        virtualScroll: true,
		        virtualScrollProps: this.tableVirtualScrollProps,
		        rowKey: this.tableRowKey,
		        pageSize: this.tablePageSize,
		        pageSizes: TABLE_PAGE_SIZES,
		        paginationProps: this.tablePaginationProps,
		        beforeLoad: this.validateTableQuery,
		        pageParam: 'current',
		        limitParam: 'size',
		        dataPath: 'data.records',
		        totalPath: 'data.total'
		      }
		    }
		  },
		  watch: {
		    'queryParams.shipTimeoutType': 'syncQuickFilters',
		    'queryParams.collectionRecordType': 'syncQuickFilters',
		    'queryParams.isTrackStalled': 'syncQuickFilters',
		    'queryParams.tagStatus': 'syncQuickFilters',
		    'queryParams.warningColorList': 'syncQuickFilters',
		    'queryParams.remarkFollowStatus': 'syncQuickFilters',
		    'queryParams.noTrackOrder': 'syncQuickFilters'
		  },
		  methods: {
		    async loadStatistics() {
		      try {
		        const response = await getLogisticsTrackingStatisticsApi()
		        const statistics = (response && response.data) || {}
		        this.statusCards = createStatisticsCards(statistics)
		        this.lastStatTime = formatStatisticsLastStatTime(statistics)
		      } catch (error) {
		        // 请求层统一提示接口错误，保留占位数据等待用户刷新页面重试。
		      }
		    },
		    handleQuickFilterChange(selectedItems) {
		      const filters = selectedItems.map(item => {
		        return item && typeof item === 'object' ? item.value : item
		      })
		      const changedFilters = ['noPickup', 'stagnant', 'none', 'red', 'followed', 'noTrackOrder'].filter(filter => {
		        return filters.includes(filter) !== this.previousQuickFilters.includes(filter)
		      })
		      changedFilters.forEach(filter => this.syncQuickFilterField(filter, filters.includes(filter)))
		      this.previousQuickFilters = [...filters]
		    },
		    syncQuickFilterField(filter, isSelected) {
		      const fieldHandlers = {
		        noPickup: () => {
		          this.queryParams.shipTimeoutType = isSelected ? 'OVER_48H' : ''
		          this.queryParams.collectionRecordType = isSelected ? 'NONE' : ''
		        },
		        stagnant: () => {
		          this.queryParams.isTrackStalled = isSelected
		        },
		        none: () => {
		          this.queryParams.tagStatus = isSelected ? 'NONE' : ''
		        },
		        red: () => {
		          this.queryParams.warningColorList = isSelected ? ['R'] : []
		        },
		        followed: () => {
		          this.queryParams.remarkFollowStatus = isSelected ? 'HAS' : ''
		        },
		        noTrackOrder: () => {
		          this.queryParams.noTrackOrder = isSelected
		        }
		      }
		      fieldHandlers[filter]()
		    },
		    syncQuickFilters() {
		      const {
		        shipTimeoutType,
		        collectionRecordType,
		        isTrackStalled,
		        tagStatus,
		        warningColorList,
		        remarkFollowStatus,
		        noTrackOrder
		      } = this.queryParams
		      const filters = [
		        shipTimeoutType === 'OVER_48H' &&
		          collectionRecordType === 'NONE' && 'noPickup',
		        isTrackStalled === true && 'stagnant',
		        tagStatus === 'NONE' && 'none',
		        Array.isArray(warningColorList) &&
		          warningColorList.length === 1 &&
		          warningColorList[0] === 'R' && 'red',
		        remarkFollowStatus === 'HAS' && 'followed',
		        noTrackOrder === true && 'noTrackOrder'
		      ].filter(Boolean)
		      this.queryParams.quickFilters = filters
		      this.previousQuickFilters = [...filters]
		    },
		    handleQuery() {
		      if (this.$refs.searchTable) this.$refs.searchTable.search()
		    },
		    validateSearch(queryParams) {
		      return this.validateQueryCondition(queryParams) &&
		        this.validateDateRanges(queryParams)
		    },
		    buildTableQuery(params) {
		      return buildLogisticsTrackingQuery(params)
		    },
		    validateQueryCondition(queryParams = this.queryParams) {
		      if (hasFilledField(queryParams, QUERY_CONDITION_FIELDS)) return true
		      this.$message.warning('查询条件不能为空，请至少填写一个查询条件')
		      return false
		    },
		    validateTableQuery() {
		      return this.validateQueryCondition()
		    },
		    validateDateRanges(queryParams = this.queryParams) {
		      const invalidRange = findInvalidDateRange(queryParams)
		      if (invalidRange) {
		        this.$message.warning(`${invalidRange.label}的开始时间不能晚于结束时间`)
		        return false
		      }
		      return true
		    },
		    handleDateRangeChange({ label, startField, endField, changedField, value }) {
		      if (value && changedField === endField) {
		        value = value.replace(/\s.*$/, ' 23:59:59')
		        this.$set(this.queryParams, changedField, value)
		      }
		      const startTime = changedField === startField
		        ? value
		        : this.queryParams[startField]
		      const endTime = changedField === endField
		        ? value
		        : this.queryParams[endField]
		      if (isStrictlyOrderedDateRange(startTime, endTime)) return
		      this.$set(this.queryParams, changedField, '')
		      this.$message.warning(`${label}的开始时间不能晚于结束时间`)
		    },
		    handleSearchReset() {
		      this.previousQuickFilters = []
		      this.queryParams.isTrackStalled = false
		      this.queryParams.noTrackOrder = false
		    },
		    handleSearch(queryParams) {
		      this.totalApi = {
		        api: getLogisticsTrackingTotalApi,
		        params: queryParams,
		        dataPath: 'data'
		      }
		    },
		    reloadTable() {
		      this.$nextTick(() => {
		        if (this.$refs.searchTable) this.$refs.searchTable.reload()
		      })
		    },
		    handleTableLoadSuccess(rows) {
		      rows.forEach(row => {
		        if (!Array.isArray(row.tags)) this.$set(row, 'tags', [])
		      })
		      this.selectedRows = []
		    },
		    handleTableLoadError() {
		      this.selectedRows = []
		    },
		    handleSelectionChange(rows) {
		      this.selectedRows = rows
		    },
		    handleTableActionClick(item, index, event, context) {
		      const { action, key } = item
		      const handlers = {
		        addTag: this.openTagDialog,
		        removeTag: this.openRemoveTagDialog,
		        importRemark: this.openBatchImportRemark,
		        remark: this.openBatchRemark,
		        export: this.handleExport,
		        trace: this.handlePullTrace
		      }
		      const handler = handlers[action || key]
		      if (handler) handler(context)
		    },
		    openTracking(row) {
		      if (!row || !row.trackNo) return
		      const url = `https://t.17track.net/zh-cn#nums=${encodeURIComponent(row.trackNo)}`
		      window.open(url, '_blank', 'noopener,noreferrer')
		    },
		    openTagDialog(context) {
		      this.$refs.tagManagementDialog.open(context)
		    },
		    openRemoveTagDialog(context) {
		      this.$refs.removeTagDialog.open(context)
		    },
		    async handleRowTagClick(row, tag) {
		      const {
		        id: tagId,
		        name = ''
		      } = tag || {}
		      const tagName = String(name).trim()
		      const idList = collectValues([row], 'id')
		      const innerOrderNos = collectValues([row], 'innerOrderNo', { strict: true })
		      const hasTagId = tagId !== null && tagId !== undefined && tagId !== ''
		      if (!idList.length || !innerOrderNos.length || (!hasTagId && !tagName)) {
		        this.$message.warning('未获取到订单或标签信息，无法移除标签')
		        return
		      }
		      try {
		        await this.$confirm(`确认移除标签“${tagName || tagId}”吗？`, '移除标签', {
		          confirmButtonText: '确认移除',
		          cancelButtonText: '取消',
		          type: 'warning'
		        })
		      } catch (error) {
		        return
		      }
		      await removeLogisticsTrackingOrderTagApi({
		        idList,
		        innerOrderNos,
		        ...(hasTagId
		          ? { tagIds: [tagId] }
		          : { tags: [tagName] })
		      })
		      this.reloadTable()
		      this.$message.success(`已移除标签“${tagName || tagId}”`)
		    },
		    openBatchImportRemark() {
		      this.$refs.remarkImportDialog.open()
		    },
		    openBatchRemark(context) {
		      this.$refs.remarkDialog.open({
		        ...context,
		        isBatch: true
		      })
		    },
		    openRemarkHistory(row) {
		      this.$refs.remarkHistoryDrawer.open(row)
		    },
		    openSingleRemarkDialog(row) {
		      this.$refs.remarkDialog.open({
		        selectedRows: [row],
		        hasSelection: true,
		        isBatch: false
		      })
		    },
		    async handleRemarkConfirm(payload) {
		      if (!payload.isBatch && this.$refs.remarkHistoryDrawer) {
		        await this.$refs.remarkHistoryDrawer.refresh()
		      }
		      this.reloadTable()
		    },
		    async handleExport() {
		      if (this.exportLoading) return
		      if (!this.validateDateRanges()) return
		      try {
		        await this.$confirm('是否确认导出物流跟踪订单？', '导出确认', {
		          confirmButtonText: '确认导出',
		          cancelButtonText: '取消',
		          type: 'warning'
		        })
		      } catch (error) {
		        return
		      }
		      this.exportLoading = true
		      try {
		        const idList = collectValues(this.selectedRows, 'id')
		        const response = await exportLogisticsTrackingApi({
		          ...buildLogisticsTrackingQuery(this.queryParams),
		          ...(idList.length ? { idList } : {})
		        })
		        // const headers = (response && response.headers) || {}
		        // const contentDisposition = headers['content-disposition'] || headers['Content-Disposition']
		        // const filename = getDownloadFilename(contentDisposition, '物流跟踪订单.xlsx')
		        // const blob = response && response.data
		        // if (!(blob instanceof Blob)) throw new Error('导出接口未返回有效文件流')
		        // this.exportFile(blob, filename, (blob && blob.type) || 'application/octet-stream')
		        if (response.success) {
		          this.$message.success('导出成功,请在下载中心查看进度！')
		        }
		      } catch (error) {
		        if (error && error.message === '导出接口未返回有效文件流') {
		          this.$message.error(error.message)
		        }
		      } finally {
		        this.exportLoading = false
		      }
		    },
		    async handlePullTrace() {
		      if (!this.canExportOrPull) {
		        this.$message.warning('请先选择要拉取物流轨迹的订单')
		        return
		      }
		      const res = await this.$confirm('是否确认拉取物流轨迹？', '拉取确认', {
		        confirmButtonText: '确认拉取',
		        cancelButtonText: '取消',
		        type: 'warning'
		      }).catch(() => {
		        return false;
		      });
		      if (!res) return;
		      const idList = collectValues(this.selectedRows, 'id')
		      await pullLogisticsTrace({
		        ...buildLogisticsTrackingQuery(this.queryParams),
		        ...(idList.length ? { idList } : {})
		      })
		      this.$message.success('已提交拉取请求，请稍后刷新查看结果')
		    }
		  }
		}
	</script>
	```
